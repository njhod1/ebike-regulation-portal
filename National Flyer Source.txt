\section*{1. UNIVERSAL RULES (ALL STATES)}
\begin{itemize}[label=--, nosep, topsep=3pt, itemsep=1pt]
  \item \textbf{250W Maximum:} Motor power must not exceed 250W on any public road or path.
  \item \textbf{25 km/h Cut-off:} Motor assistance must stop at 25 km/h in all jurisdictions.
  \item \textbf{No Throttle Cruising:} Throttle-only riding above 6 km/h is illegal nationwide.
  \item \textbf{EN 15194 Standard:} Mandatory on all new e-bikes sold in NSW from Feb 2026; a national import standard applies from Dec 2025. Road-use compliance deadlines vary by state — see table below.
  \item \textbf{Over 250W = Motorcycle:} Requires registration, insurance and a licence. Riding without these is a criminal offence.
  \item \textbf{Helmet Required:} An approved bicycle helmet is mandatory in all states.
\end{itemize}

\section*{3. PARENTS: IS YOUR HOME AT RISK?}
\begin{tcolorbox}[colback=white, colframe=DangerRed, top=3pt, bottom=3pt]
  \small \textbf{If an underage rider causes an accident on an illegal e-bike:}
  \hyphenpenalty=10000 \exhyphenpenalty=10000
  \begin{enumerate}[nosep, topsep=2pt, itemsep=1pt]
    \item \textbf{NO INSURANCE:} Home policies exclude ``unregistered motor vehicles.'' Illegal e-bikes fall under this category.

    \item \textbf{ASSET LIQUIDATION:} Your house, savings and assets can be seized to pay legal debts.
